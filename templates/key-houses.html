Four standards, different jobs.

PKCS #1
RSA key formats and algorithms. It is not a certificate suitcase.
PKCS #8
A private-key container. This lab encrypts it with your password.
PKCS #11
An interface to cryptographic tokens, such as smart cards. Not implemented here.
PKCS #12
A password-protected bundle for a private key and certificate.

What this demo does not prove.

A self-signed certificate is not trusted identity evidence. Check its fingerprint and expiry, but do not use it to identify a real person or service.

Signing a challenge verifies a key match. This is not a physical lock, a smart card or production authentication.

Lab-only browser storage. This browser saves encrypted files, public keys and certificate metadata, never the password or decrypted key. Same-origin scripts can read or copy those encrypted files; offline password guessing is possible. Use disposable passwords. Clear the demo when finished. Clearing here does not delete downloaded copies or server files.

def packaging(p, state):
    # Only encrypted laboratory artifacts travel to/from browser storage.
    if p.get('action') == 'open':
        password = text(p, 'password', limit=128).encode('utf-8')
        artifact = text(p, 'artifact', limit=16000)
        try:
            if p.get('format') == 'pkcs8':
                private = serialization.load_pem_private_key(artifact.encode('ascii'), password)
                certificate_matches = None
            elif p.get('format') == 'pkcs12':
                private, certificate, _ = pkcs12.load_key_and_certificates(
                    base64.b64decode(artifact, validate=True), password)
                if private is None or certificate is None:
                    raise ValueError('Missing key or certificate.')
                certificate_matches = private.public_key().public_numbers() == certificate.public_key().public_numbers()
            else:
                raise ValueError('Unknown container.')
            if not isinstance(private, rsa.RSAPrivateKey):
                raise ValueError('Expected an RSA laboratory key.')
        except (ValueError, TypeError, UnicodeError):
            raise ValueError('Incorrect password or invalid encrypted package.') from None
        try:
            public = serialization.load_pem_public_key(text(p, 'public_key', limit=2000).encode('ascii'))
            if not isinstance(public, rsa.RSAPublicKey):
                raise ValueError('Expected an RSA public key.')
        except (ValueError, TypeError, UnicodeError):
            raise ValueError('Invalid house public key.') from None
        # Actual signature verification, not a browser-side house-ID comparison.
        challenge = secrets.token_bytes(32)
        signature = private.sign(challenge, padding.PSS(mgf=padding.MGF1(hashes.SHA256()),
                                 salt_length=padding.PSS.MAX_LENGTH), hashes.SHA256())
        return {'opened': verifies(public, signature, challenge), 'key_loaded': True,
                'certificate_matches': certificate_matches,
                'operation': 'RSA-PSS challenge verification; educational key-match demo only'}
    password = text(p, 'password', limit=128)
    if len(password) < 12:
        raise ValueError('Use a disposable laboratory password of 12-128 characters.')
    secret = password.encode('utf-8')
    private = keypair()
    name = x509.Name([x509.NameAttribute(NameOID.COMMON_NAME, 'ENIGMA disposable laboratory certificate')])
    now = datetime.now(timezone.utc)
    cert = (x509.CertificateBuilder().subject_name(name).issuer_name(name).public_key(private.public_key())
            .serial_number(x509.random_serial_number()).not_valid_before(now - timedelta(minutes=1))
            .not_valid_after(now + timedelta(days=7)).add_extension(x509.BasicConstraints(ca=False, path_length=None), critical=True)
            .sign(private, hashes.SHA256()))
    pem = private.private_bytes(serialization.Encoding.PEM, serialization.PrivateFormat.PKCS8, serialization.BestAvailableEncryption(secret))
    bundle = pkcs12.serialize_key_and_certificates(b'enigma-lab', private, cert, None, serialization.BestAvailableEncryption(secret))
    restored = serialization.load_pem_private_key(pem, password=secret)
    restored12, cert12, _ = pkcs12.load_key_and_certificates(bundle, secret)
    wrong = b'deliberately-wrong-' + secrets.token_hex(16).encode('ascii')
    failures = []
    for loader, value in [(serialization.load_pem_private_key, pem), (pkcs12.load_key_and_certificates, bundle)]:
        try:
            loader(value, wrong)
            failures.append(False)
        except ValueError:
            failures.append(True)
    identifier = secrets.token_hex(16)
    folder = ROOT / state['id']
    folder.mkdir(parents=True, exist_ok=True)
    files = {'private-key.pem': pem, 'bundle.p12': bundle, 'certificate.pem': cert.public_bytes(serialization.Encoding.PEM)}
    links = {}
    for filename, data in files.items():
        path = folder / (identifier + '-' + filename)
        path.write_bytes(data)
        token = secrets.token_urlsafe(24)
        state['downloads'][token] = (path, filename)
        links[filename] = '/api/download/' + token
    public = private.public_key().public_numbers()
    house_public_keys = [key.public_bytes(serialization.Encoding.PEM,
                         serialization.PublicFormat.SubjectPublicKeyInfo).decode('ascii')
                         for key in (private.public_key(), keypair().public_key(), keypair().public_key())]
    # Carry encrypted files in the generation response: cloud workers do not share
    # in-memory download tokens or local files. Never include an unencrypted key.
    artifacts = {'pkcs8': pem.decode('ascii'), 'pkcs12': base64.b64encode(bundle).decode('ascii'),
                 'certificate': cert.public_bytes(serialization.Encoding.PEM).decode('ascii')}
    return {'artifacts': artifacts, 'house_public_keys': house_public_keys, 'downloads': links, 'pkcs8_recovered': restored.public_key().public_numbers() == public,
            'pkcs12_recovered': restored12.public_key().public_numbers() == public and cert12.fingerprint(hashes.SHA256()) == cert.fingerprint(hashes.SHA256()),
            'wrong_password_rejected_pkcs8': failures[0], 'wrong_password_rejected_pkcs12': failures[1],
            'certificate_sha256': cert.fingerprint(hashes.SHA256()).hex(), 'valid_until': cert.not_valid_after_utc.isoformat(),
            'standards': {'PKCS #1': 'RSA key format and algorithms', 'PKCS #8': 'Private-key storage; encrypted here', 'PKCS #11': 'Interface for cryptographic tokens; not implemented by this simulation', 'PKCS #12': 'Password-protected key and certificate bundle'},
            'warning': 'Self-signed test certificate: not trusted identity evidence. Only encrypted private-key artifacts are saved. Keep your disposable password locally; it is not saved. Downloads require this browser session. Server restart discards access links.'}
